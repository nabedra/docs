```python Python
from nabedra import Database, col

db = Database("http://127.0.0.1:8000", mode="olap")
tasks = db.create_table(
    "tasks",
    {"id": str, "title": str, "priority": int},
    key="id",
)
tasks.insert_many([
    {"id": "t1", "title": "Write docs", "priority": 4},
])

urgent = tasks.where(col("priority") >= 3)
print(urgent.select("id", "title").collect())
```

```typescript TypeScript
import { Database, col } from "nabedra"

const db = new Database({ url: "http://127.0.0.1:8000" })
const users = await db.createTable(
  "users",
  { id: String, age: BigInt },
  "id",
)
await users.insertMany([
  { id: "alice", age: 24n },
])

const adults = users.where(col("age").ge(18n)).select("id", "age")
console.log(await adults.collect())
```

```python
from nabedra.cache import Cache

with Cache.connect("redis://127.0.0.1:6379", namespace="catalog") as cache:
    cache.set("sku:42", b"in stock", ttl_ms=30_000)
    value = cache.get("sku:42")
    print("cache miss" if value is None else value.decode("utf-8"))
```

```python
import os
from uuid import uuid4

from nabedra.events import Events

producer = Events(
    "http://127.0.0.1:18080",
    token=os.environ["NABEDRA_EVENTS_PUBLISH_TOKEN"],
    allow_insecure=True,
)
published = producer.publish(
    "orders", partition=0, event_id=uuid4(), data={"order_id": "A-42"},
)
print(published.offset)
```

<div>

```python
from nabedra import Database, col

db = Database("https://nabedra.example", database="ml", mode="olap")
query = (
    db.table("training_examples")
    .query()
    .where((col("split") == "train") & (col("weight") > 0.0))
    .select("x1", "x2", "label")
)

with query.stream(batch_rows=64) as batches:
    for rows in batches:
        print(rows)
```
